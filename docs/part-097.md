## 1. Source Generator คืออะไร

Source Generator เป็น compiler extension ที่รัน during compilation และสร้าง C# code ได้โดยอัตโนมัติ

### ประโยชน์
- ลด boilerplate code
- Zero-cost abstraction (code generate ที่ compile time)
- ดีกว่า reflection (no runtime overhead)
- ดีกว่า T4 templates (integrate กับ Roslyn)

### ตัวอย่างที่ใช้ใน .NET
- `System.Text.Json` → JsonSerializerContext
- `Regex` → `[GeneratedRegex]`
- `LoggerMessage` → `[LoggerMessage]`
- `EF Core` → Compiled models

## 3. Incremental Source Generator (แนะนำ)

```csharp
// SourceGenerators/NotifyPropertyChangedGenerator.cs
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using System.Collections.Immutable;
using System.Text;

[Generator]
public class NotifyPropertyChangedGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        // Step 1: Find all classes with [GenerateNotify] attribute
        var classDeclarations = context.SyntaxProvider
            .ForAttributeWithMetadataName(
                "GenerateNotifyAttribute",
                predicate: static (node, _) => node is ClassDeclarationSyntax,
                transform: static (ctx, _) => GetClassInfo(ctx))
            .Where(static info => info is not null)
            .Select(static (info, _) => info!);

        // Step 2: Combine with compilation
        var compilationAndClasses = context.CompilationProvider.Combine(classDeclarations.Collect());

        // Step 3: Generate code
        context.RegisterSourceOutput(compilationAndClasses, 
            static (spc, source) => Execute(source.Left, source.Right, spc));
    }

    private static ClassInfo? GetClassInfo(GeneratorAttributeSyntaxContext ctx)
    {
        if (ctx.TargetSymbol is not INamedTypeSymbol classSymbol)
            return null;

        var properties = classSymbol.GetMembers()
            .OfType<IPropertySymbol>()
            .Where(p => p.GetAttributes().Any(a => a.AttributeClass?.Name == "NotifyAttribute"))
            .Select(p => new PropertyInfo(p.Name, p.Type.ToDisplayString()))
            .ToList();

        return new ClassInfo(
            classSymbol.Name,
            classSymbol.ContainingNamespace.ToDisplayString(),
            properties);
    }

    private static void Execute(
        Compilation compilation,
        ImmutableArray<ClassInfo> classes,
        SourceProductionContext ctx)
    {
        foreach (var classInfo in classes)
        {
            var source = GenerateClass(classInfo);
            ctx.AddSource($"{classInfo.ClassName}.g.cs", source);
        }
    }

    private static string GenerateClass(ClassInfo classInfo)
    {
        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated />");
        sb.AppendLine("#nullable enable");
        sb.AppendLine();
        sb.AppendLine($"namespace {classInfo.Namespace};");
        sb.AppendLine();
        sb.AppendLine($"partial class {classInfo.ClassName} : System.ComponentModel.INotifyPropertyChanged");
        sb.AppendLine("{");
        sb.AppendLine("    public event System.ComponentModel.PropertyChangedEventHandler? PropertyChanged;");
        sb.AppendLine();
        sb.AppendLine("    protected virtual void OnPropertyChanged(string propertyName)");
        sb.AppendLine("        => PropertyChanged?.Invoke(this, new System.ComponentModel.PropertyChangedEventArgs(propertyName));");
        sb.AppendLine();

        foreach (var prop in classInfo.Properties)
        {
            var fieldName = $"_{char.ToLower(prop.Name[0])}{prop.Name[1..]}";
            sb.AppendLine($"    private {prop.TypeName} {fieldName};");
            sb.AppendLine();
            sb.AppendLine($"    public {prop.TypeName} {prop.Name}");
            sb.AppendLine("    {");
            sb.AppendLine($"        get => {fieldName};");
            sb.AppendLine("        set");
            sb.AppendLine("        {");
            sb.AppendLine($"            if ({fieldName} != value)");
            sb.AppendLine("            {");
            sb.AppendLine($"                {fieldName} = value;");
            sb.AppendLine($"                OnPropertyChanged(nameof({prop.Name}));");
            sb.AppendLine("            }");
            sb.AppendLine("        }");
            sb.AppendLine("    }");
            sb.AppendLine();
        }

        sb.AppendLine("}");
        return sb.ToString();
    }
}

public record ClassInfo(string ClassName, string Namespace, List<PropertyInfo> Properties);
public record PropertyInfo(string Name, string TypeName);
```

[Generator]
public class AttributeGenerator : IIncrementalGenerator
{
    private const string GenerateNotifyAttribute = """
        // <auto-generated />
        namespace System
        {
            [AttributeUsage(AttributeTargets.Class)]
            internal sealed class GenerateNotifyAttribute : Attribute { }
            
            [AttributeUsage(AttributeTargets.Property)]
            internal sealed class NotifyAttribute : Attribute { }
        }
        """;

    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        context.RegisterPostInitializationOutput(ctx =>
            ctx.AddSource("GenerateNotifyAttribute.g.cs", GenerateNotifyAttribute));
    }
}

## 6. Strongly-typed Options Generator

```csharp
// Generator สำหรับสร้าง strongly-typed options classes
[Generator]
public class StronglyTypedOptionsGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        // อ่านจาก AdditionalFiles (appsettings.json)
        var jsonFiles = context.AdditionalTextsProvider
            .Where(f => f.Path.EndsWith("appsettings.json", StringComparison.OrdinalIgnoreCase));

        context.RegisterSourceOutput(jsonFiles, static (spc, file) =>
        {
            var content = file.GetText()?.ToString();
            if (content == null) return;

            try
            {
                var options = ParseOptions(content);
                var source = GenerateOptionsClasses(options);
                spc.AddSource("GeneratedOptions.g.cs", source);
            }
            catch (Exception ex)
            {
                spc.ReportDiagnostic(Diagnostic.Create(
                    new DiagnosticDescriptor("OPT001", "Options error", ex.Message,
                        "Options", DiagnosticSeverity.Warning, true),
                    Location.None));
            }
        });
    }

    private static Dictionary<string, object> ParseOptions(string json)
    {
        using var doc = System.Text.Json.JsonDocument.Parse(json);
        return doc.RootElement.EnumerateObject()
            .ToDictionary(
                p => p.Name,
                p => (object)p.Value.ToString());
    }

    private static string GenerateOptionsClasses(Dictionary<string, object> sections)
    {
        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated from appsettings.json />");
        sb.AppendLine("namespace GeneratedOptions;");
        sb.AppendLine();

        foreach (var (section, _) in sections)
        {
            sb.AppendLine($"/// <summary>Strongly-typed options for {section} section</summary>");
            sb.AppendLine($"public partial class {section}Options");
            sb.AppendLine("{");
            sb.AppendLine($"    public const string SectionName = \"{section}\";");
            sb.AppendLine("}");
            sb.AppendLine();
        }

        return sb.ToString();
    }
}
```

## 7. Roslyn Analyzers

Analyzers ตรวจสอบ code ที่ compile time และ report diagnostics

```csharp
// Analyzers/SealedClassAnalyzer.cs
[DiagnosticAnalyzer(LanguageNames.CSharp)]
public class SealedClassAnalyzer : DiagnosticAnalyzer
{
    public static readonly DiagnosticDescriptor Rule = new(
        id: "CA9001",
        title: "Record class should be sealed",
        messageFormat: "Record class '{0}' should be sealed to prevent inheritance issues",
        category: "Design",
        defaultSeverity: DiagnosticSeverity.Warning,
        isEnabledByDefault: true,
        description: "Non-sealed record classes can cause equality issues with inheritance.");

    public override ImmutableArray<DiagnosticDescriptor> SupportedDiagnostics 
        => ImmutableArray.Create(Rule);

    public override void Initialize(AnalysisContext context)
    {
        context.ConfigureGeneratedCodeAnalysis(GeneratedCodeAnalysisFlags.None);
        context.EnableConcurrentExecution();
        context.RegisterSyntaxNodeAction(AnalyzeNode, SyntaxKind.RecordDeclaration);
    }

    private static void AnalyzeNode(SyntaxNodeAnalysisContext context)
    {
        var recordDeclaration = (RecordDeclarationSyntax)context.Node;
        
        // Check if record class (not struct) and not sealed
        if (recordDeclaration.ClassOrStructKeyword.IsKind(SyntaxKind.ClassKeyword) &&
            !recordDeclaration.Modifiers.Any(SyntaxKind.SealedKeyword) &&
            !recordDeclaration.Modifiers.Any(SyntaxKind.AbstractKeyword))
        {
            var diagnostic = Diagnostic.Create(
                Rule,
                recordDeclaration.Identifier.GetLocation(),
                recordDeclaration.Identifier.Text);
            
            context.ReportDiagnostic(diagnostic);
        }
    }
}

// Code Fix สำหรับ Analyzer
[ExportCodeFixProvider(LanguageNames.CSharp, Name = nameof(SealedClassCodeFix))]
[Shared]
public class SealedClassCodeFix : CodeFixProvider
{
    public override ImmutableArray<string> FixableDiagnosticIds 
        => ImmutableArray.Create(SealedClassAnalyzer.Rule.Id);

    public override FixAllProvider GetFixAllProvider() => WellKnownFixAllProviders.BatchFixer;

    public override async Task RegisterCodeFixesAsync(CodeFixContext context)
    {
        var root = await context.Document.GetSyntaxRootAsync(context.CancellationToken);
        var diagnostic = context.Diagnostics.First();
        var node = root?.FindNode(diagnostic.Location.SourceSpan);
        
        if (node is RecordDeclarationSyntax recordDecl)
        {
            context.RegisterCodeFix(
                CodeAction.Create(
                    title: "Add 'sealed' modifier",
                    createChangedDocument: ct => AddSealedModifier(context.Document, recordDecl, ct),
                    equivalenceKey: "AddSealed"),
                diagnostic);
        }
    }

    private static async Task<Document> AddSealedModifier(
        Document document,
        RecordDeclarationSyntax recordDecl,
        CancellationToken ct)
    {
        var sealedToken = SyntaxFactory.Token(SyntaxKind.SealedKeyword)
            .WithTrailingTrivia(SyntaxFactory.Space);

        var newModifiers = recordDecl.Modifiers.Insert(0, sealedToken);
        var newDecl = recordDecl.WithModifiers(newModifiers);

        var root = await document.GetSyntaxRootAsync(ct);
        var newRoot = root!.ReplaceNode(recordDecl, newDecl);
        return document.WithSyntaxRoot(newRoot);
    }
}
```

## 8. Testing Source Generators

```csharp
// Tests/NotifyGeneratorTests.cs
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Xunit;

public class NotifyGeneratorTests
{
    [Fact]
    public void Generator_Should_Generate_INotifyPropertyChanged()
    {
        // Arrange
        var source = """
            using System;
            
            [GenerateNotify]
            public partial class PersonViewModel
            {
                [Notify]
                public string Name { get; set; } = string.Empty;
            }
            """;

        // Act
        var (compilation, diagnostics) = RunGenerator(source);

        // Assert
        Assert.Empty(diagnostics.Where(d => d.Severity == DiagnosticSeverity.Error));
        
        var generatedTree = compilation.SyntaxTrees
            .FirstOrDefault(t => t.FilePath.EndsWith(".g.cs"));
        Assert.NotNull(generatedTree);
        
        var generatedText = generatedTree.ToString();
        Assert.Contains("INotifyPropertyChanged", generatedText);
        Assert.Contains("OnPropertyChanged", generatedText);
        Assert.Contains("PropertyChanged", generatedText);
    }

    [Fact]
    public void Generator_Should_Generate_Backing_Field()
    {
        var source = """
            [GenerateNotify]
            public partial class TestViewModel
            {
                [Notify]
                public int Count { get; set; }
            }
            """;

        var (compilation, _) = RunGenerator(source);
        
        var generatedCode = compilation.SyntaxTrees
            .FirstOrDefault(t => t.FilePath.EndsWith(".g.cs"))
            ?.ToString() ?? "";

        Assert.Contains("private int _count", generatedCode);
        Assert.Contains("get => _count", generatedCode);
    }

    private static (Compilation, ImmutableArray<Diagnostic>) RunGenerator(string source)
    {
        var inputCompilation = CSharpCompilation.Create(
            "TestAssembly",
            [CSharpSyntaxTree.ParseText(source)],
            [MetadataReference.CreateFromFile(typeof(object).Assembly.Location)],
            new CSharpCompilationOptions(OutputKind.DynamicallyLinkedLibrary));

        var generator = new NotifyPropertyChangedGenerator();
        
        CSharpGeneratorDriver
            .Create(generator)
            .RunGeneratorsAndUpdateCompilation(
                inputCompilation, out var outputCompilation, out var diagnostics);

        return (outputCompilation, diagnostics);
    }
}
```

## 9. Real-world: Mapper Generator

```csharp
// ใช้ [GenerateMapper] attribute เพื่อสร้าง mapping code
[Generator]
public class MapperGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        var mappers = context.SyntaxProvider
            .ForAttributeWithMetadataName(
                "GenerateMapperAttribute",
                predicate: (n, _) => n is ClassDeclarationSyntax,
                transform: GetMapperInfo)
            .Where(x => x is not null);

        context.RegisterSourceOutput(mappers, static (spc, info) =>
        {
            var source = GenerateMapper(info!);
            spc.AddSource($"{info!.ClassName}Mapper.g.cs", source);
        });
    }

    private static MapperInfo? GetMapperInfo(GeneratorAttributeSyntaxContext ctx, CancellationToken ct)
    {
        if (ctx.TargetSymbol is not INamedTypeSymbol classSymbol) return null;
        
        var attr = ctx.Attributes.FirstOrDefault();
        if (attr?.ConstructorArguments.FirstOrDefault().Value is not INamedTypeSymbol targetType)
            return null;

        var sourceProps = classSymbol.GetMembers().OfType<IPropertySymbol>().ToList();
        var targetProps = targetType.GetMembers().OfType<IPropertySymbol>().ToList();

        var mappings = sourceProps
            .Join(targetProps, s => s.Name, t => t.Name, (s, t) => new PropertyMapping(
                s.Name, s.Type.ToDisplayString(), t.Type.ToDisplayString()))
            .ToList();

        return new MapperInfo(classSymbol.Name, targetType.Name, 
            classSymbol.ContainingNamespace.ToDisplayString(), mappings);
    }

    private static string GenerateMapper(MapperInfo info)
    {
        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated />");
        sb.AppendLine($"namespace {info.Namespace};");
        sb.AppendLine();
        sb.AppendLine($"public static partial class {info.ClassName}Mapper");
        sb.AppendLine("{");
        sb.AppendLine($"    public static {info.TargetClassName} MapTo{info.TargetClassName}(this {info.ClassName} source)");
        sb.AppendLine("    {");
        sb.AppendLine($"        return new {info.TargetClassName}");
        sb.AppendLine("        {");

        foreach (var mapping in info.Mappings)
        {
            sb.AppendLine($"            {mapping.PropertyName} = source.{mapping.PropertyName},");
        }

        sb.AppendLine("        };");
        sb.AppendLine("    }");
        sb.AppendLine("}");
        return sb.ToString();
    }
}

public record MapperInfo(
    string ClassName, 
    string TargetClassName, 
    string Namespace, 
    List<PropertyMapping> Mappings);

public record PropertyMapping(string PropertyName, string SourceType, string TargetType);

// ใช้งาน
[GenerateMapper(typeof(ProductDto))]
public partial class Product
{
    public Guid Id { get; set; }
    public string Name { get; set; } = string.Empty;
    public decimal Price { get; set; }
}

public class ProductDto
{
    public Guid Id { get; set; }
    public string Name { get; set; } = string.Empty;
    public decimal Price { get; set; }
}

// Generated code:
// public static partial class ProductMapper
// {
//     public static ProductDto MapToProductDto(this Product source) => new ProductDto { ... };
// }

// ใช้ใน code:
var product = new Product { Id = Guid.NewGuid(), Name = "Test", Price = 100 };
var dto = product.MapToProductDto();  // Generated!
```

## สรุป

- **IIncrementalGenerator** ดีกว่า ISourceGenerator (incremental, faster)
- **SyntaxProvider** ค้นหา syntax nodes ที่ต้องการ
- **PostInitializationOutput** ใช้สำหรับเพิ่ม attribute definitions
- **Diagnostics** report errors/warnings จาก generator
- **Code Fixes** เพิ่ม automatic fixes ให้ analyzer
- **Testing** ใช้ CSharpGeneratorDriver เพื่อ test generators
